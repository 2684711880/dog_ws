# 四足机器人运动控制 完全重构方案

## 一、当前代码的根本问题（不是修补能解决的）

1. **没有腿坐标系** — 四条腿共享 `stand_foot_x/y`，前后腿 hip offset 不同但代码不区分
2. **开环执行** — 假设上一步完美执行，没有足端实际位置反馈
3. **planner 和 runtime 职责混乱** — 滤波、相位推进、轨迹生成、IK 搅在一起
4. **没有足端工作空间** — 靠运行时 IK 失败来兜底
5. **支撑相速度曲线错误** — smoothstep 不匹配匀速体运动
6. **duty factor 硬编码 50%** — trot 无飞行相容忍度

## 三、核心数据结构

```cpp
// ======================== types.h ========================

#pragma once
#include <array>
#include <cmath>
#include <cstddef>
#include <string>

namespace quad {

// ---- 常量 ----
constexpr size_t kNumLegs = 4;
constexpr size_t kJointsPerLeg = 3;
constexpr size_t kNumJoints = kNumLegs * kJointsPerLeg;
constexpr double kPI = 3.14159265358979323846;
constexpr double k2PI = 2.0 * kPI;

// ---- 向量 ----
struct Vec3 {
    double x{0}, y{0}, z{0};
    Vec3() = default;
    Vec3(double x_, double y_, double z_) : x(x_), y(y_), z(z_) {}
    Vec3 operator+(const Vec3& o) const { return {x+o.x, y+o.y, z+o.z}; }
    Vec3 operator-(const Vec3& o) const { return {x-o.x, y-o.y, z-o.z}; }
    Vec3 operator*(double s) const { return {x*s, y*s, z*s}; }
    double norm() const { return std::sqrt(x*x + y*y + z*z); }
};

// ---- 腿和关节 ID ----
enum class LegID : size_t { LF=0, RF=1, LB=2, RB=3 };
enum class JointID : size_t { HIP_YAW=0, HIP_PITCH=1, KNEE=2 };

// ---- 单条腿的关节角 ----
struct JointAngles {
    double hip_yaw{0};
    double hip_pitch{0};
    double knee{0};

    double& operator[](size_t i) {
        switch(i) { case 0: return hip_yaw; case 1: return hip_pitch; default: return knee; }
    }
    const double& operator[](size_t i) const {
        switch(i) { case 0: return hip_yaw; case 1: return hip_pitch; default: return knee; }
    }
};

// ---- 单条腿的状态 ----
struct LegState {
    JointAngles  joint_pos;         // 当前关节角 (rad)
    JointAngles  joint_vel;         // 当前关节速度 (rad/s)
    Vec3         foot_pos_hip;      // 足端在髋关节坐标系下的位置 (FK算出)
    Vec3         foot_pos_body;     // 足端在体坐标系下的位置
    bool         in_contact{true};  // 是否触地 (如果有足底力传感器)
};

// ---- 全机器人状态 ----
struct RobotState {
    std::array<LegState, kNumLegs> legs;
    Vec3 body_rpy{};               // roll, pitch, yaw (来自IMU，没有就全0)
    Vec3 body_angular_vel{};
};

// ---- 运动指令 ----
struct LocomotionCommand {
    double vx{0};                  // 前进速度 m/s
    double vy{0};                  // 侧移速度 m/s
    double yaw_rate{0};            // 转向角速度 rad/s
    double body_height{0.29};      // 期望站立高度 m
};

// ---- 步态类型 ----
enum class GaitType : uint8_t {
    STAND = 0,
    TROT  = 1,
    WALK  = 2,
    PACE  = 3,
    BOUND = 4,
};

// ---- 步态参数 ----
struct GaitParams {
    GaitType type{GaitType::STAND};
    double   frequency{1.5};       // 步频 Hz
    double   duty_factor{0.6};     // 支撑相占比 (>0.5!)
    double   step_height{0.05};    // 抬腿高度 m
    double   max_step_length{0.14};// 最大步长 m

    // 每条腿的相位偏移 [0, 2π)
    std::array<double, kNumLegs> phase_offsets{0, kPI, kPI, 0};
    // 默认 trot: LF=0, RF=π, LB=π, RB=0
};

// ---- 接触计划 (每条腿当前是摆动还是支撑) ----
struct ContactSchedule {
    std::array<bool, kNumLegs>   in_stance;      // true = 支撑
    std::array<double, kNumLegs> phase_in_cycle;  // 每条腿自己的归一化相位 [0,1)
    std::array<double, kNumLegs> swing_progress;  // 摆动相进度 [0,1]，支撑相时为-1
    std::array<double, kNumLegs> stance_progress; // 支撑相进度 [0,1]，摆动相时为-1
    double global_phase;                           // 全局相位 [0, 2π)
};

// ---- 足端目标 (planner 输出) ----
struct FootTarget {
    Vec3   position;       // 髋关节坐标系下的目标足端位置
    bool   force_contact;  // 是否强制触地 (支撑相)
};

// ---- 关节指令 (最终输出) ----
struct JointCommand {
    std::array<JointAngles, kNumLegs> positions;
    // 后续可扩展: velocities, torques, kp, kd
};

}  // namespace quad
```

## 四、模块 1: 腿运动学 (LegKinematics)

**核心改变：每条腿有独立的 hip frame。**

```cpp
// ======================== leg_kinematics.h ========================

#pragma once
#include "types.h"

namespace quad {

// 每条腿的几何配置
struct LegConfig {
    Vec3   hip_offset_in_body;  // 髋关节在体坐标系下的位置 (固定)
    double thigh_length;
    double shank_length;
    double knee_gear_ratio;     // 你的减速比，之前硬编码的 2.0
    int    side_sign;           // +1 左腿, -1 右腿
    JointAngles joint_min;      // 关节下限
    JointAngles joint_max;      // 关节上限
};

class LegKinematics {
public:
    // 构造时传入4条腿的配置
    explicit LegKinematics(const std::array<LegConfig, kNumLegs>& configs);

    // ---- 正运动学: 关节角 → 足端位置 (髋关节坐标系) ----
    Vec3 forward_kinematics(LegID leg, const JointAngles& q) const;

    // ---- 正运动学: 关节角 → 足端位置 (体坐标系) ----
    Vec3 forward_kinematics_body(LegID leg, const JointAngles& q) const;

    // ---- 逆运动学: 足端位置 (髋关节坐标系) → 关节角 ----
    // 成功返回true, 结果写入 q_out
    bool inverse_kinematics(LegID leg, const Vec3& foot_pos_hip,
                            JointAngles& q_out) const;

    // ---- 逆运动学 (体坐标系) ----
    bool inverse_kinematics_body(LegID leg, const Vec3& foot_pos_body,
                                  JointAngles& q_out) const;

    // ---- 工作空间查询 ----
    // 判断一个足端位置是否在可达域内 (快速版，不解IK)
    bool is_reachable(LegID leg, const Vec3& foot_pos_hip) const;

    // 获取可达域边界 (用于clamp)
    // 给定足端 xy，返回z的可达范围
    bool z_reach_range(LegID leg, double x, double y,
                       double& z_min, double& z_max) const;

    // 将足端位置 clamp 到可达域内最近点
    Vec3 clamp_to_workspace(LegID leg, const Vec3& foot_pos_hip) const;

    // ---- 坐标变换 ----
    Vec3 hip_to_body(LegID leg, const Vec3& p_hip) const;
    Vec3 body_to_hip(LegID leg, const Vec3& p_body) const;

    // ---- 获取配置 ----
    const LegConfig& config(LegID leg) const;

private:
    std::array<LegConfig, kNumLegs> configs_;
};

}  // namespace quad
```

**关键改动点：**
- `hip_offset_in_body` — 你必须量出前后腿的 hip 到 body center 的 xyz 偏移
- 所有 IK/FK 输入输出都明确标注坐标系
- `clamp_to_workspace` 替代你原来的 `continue`

**实现核心逻辑：**

```cpp
ContactSchedule GaitScheduler::step(double dt) {
    // 推进全局相位
    if (params_.type != GaitType::STAND) {
        global_phase_ += k2PI * params_.frequency * dt;
        if (global_phase_ >= k2PI) global_phase_ -= k2PI;
    }

    ContactSchedule cs{};
    cs.global_phase = global_phase_;

    for (size_t i = 0; i < kNumLegs; ++i) {
        // 每条腿的局部相位
        double leg_phase = std::fmod(global_phase_ + params_.phase_offsets[i], k2PI);
        if (leg_phase < 0) leg_phase += k2PI;

        double u = leg_phase / k2PI;  // 归一化到 [0, 1)
        cs.phase_in_cycle[i] = u;

        if (params_.type == GaitType::STAND) {
            cs.in_stance[i] = true;
            cs.stance_progress[i] = 0.5;
            cs.swing_progress[i] = -1;
        } else {
            // duty_factor 部分是支撑相，剩余是摆动相
            cs.in_stance[i] = (u < params_.duty_factor);

            if (cs.in_stance[i]) {
                cs.stance_progress[i] = u / params_.duty_factor;
                cs.swing_progress[i] = -1;
            } else {
                cs.stance_progress[i] = -1;
                cs.swing_progress[i] = (u - params_.duty_factor) / (1.0 - params_.duty_factor);
            }
        }
    }

    return cs;
}
```

## 六、模块 3: 足端轨迹规划 (FootPlanner)

**核心改变：摆动相用当前实际足端位置做起点，支撑相线性。**

```cpp
// ======================== foot_planner.h ========================

#pragma once
#include "types.h"
#include "leg_kinematics.h"
#include "gait_scheduler.h"

namespace quad {

class FootPlanner {
public:
    explicit FootPlanner(const LegKinematics& kinematics);

    // 每个控制周期调用一次
    // 输入: 机器人状态 + 运动指令 + 接触计划
    // 输出: 4条腿的足端目标位置 (体坐标系)
    std::array<FootTarget, kNumLegs> plan(
        const RobotState& state,
        const LocomotionCommand& cmd,
        const ContactSchedule& schedule);

private:
    // ---- 摆动相轨迹 (Bezier曲线) ----
    Vec3 swing_trajectory(
        const Vec3& p_start,        // 摆动起点 (上一次离地位置)
        const Vec3& p_target,       // 摆动终点 (下一个落脚点)
        double swing_progress,      // [0, 1]
        double step_height) const;

    // ---- 支撑相轨迹 (线性后推) ----
    Vec3 stance_trajectory(
        const Vec3& p_touchdown,    // 这次触地时的足端位置
        double stance_progress,     // [0, 1]
        double vx, double vy,       // 体速度
        double dt_stance) const;    // 整个支撑相时长

    // ---- 计算落脚点 (Raibert heuristic) ----
    Vec3 compute_landing_target(
        LegID leg,
        const Vec3& hip_pos_body,   // 髋关节在体坐标系下的位置
        const LocomotionCommand& cmd,
        const GaitParams& gait_params) const;

private:
    const LegKinematics& kin_;

    // 每条腿记住的状态
    struct LegMemory {
        Vec3  liftoff_pos{};        // 离地时足端位置
        Vec3  touchdown_pos{};      // 触地时足端位置
        Vec3  landing_target{};     // 计划的下一个落脚点
        bool  was_in_stance{true};  // 上一个周期是否在支撑
    };
    std::array<LegMemory, kNumLegs> memory_;
};

}  // namespace quad
```

**落脚点计算 — Raibert Heuristic:**

这是四足机器人最经典的落脚点选择方法。

```cpp
Vec3 FootPlanner::compute_landing_target(
    LegID leg,
    const Vec3& hip_pos_body,
    const LocomotionCommand& cmd,
    const GaitParams& params) const
{
    const double T_stance = params.duty_factor / params.frequency;

    // Raibert: 落脚点 = 髋关节正下方 + 速度前馈 + 对称补偿
    //
    // p_landing = p_hip + v * T_stance/2 + k * (v - v_desired)
    //
    // 简化版 (没有速度反馈时):
    // p_landing = p_hip + v_cmd * T_stance / 2

    Vec3 target;
    target.x = hip_pos_body.x + cmd.vx * T_stance * 0.5;
    target.y = hip_pos_body.y + cmd.vy * T_stance * 0.5;
    target.z = -cmd.body_height;  // 地面高度

    // 转向: 左右腿落脚点左右对称偏移
    const auto& cfg = kin_.config(leg);
    double yaw_offset = cmd.yaw_rate * T_stance * 0.5;
    target.x += -cfg.hip_offset_in_body.y * std::sin(yaw_offset);
    target.y += cfg.hip_offset_in_body.y * (std::cos(yaw_offset) - 1.0);

    // Clamp 到工作空间
    Vec3 target_hip = kin_.body_to_hip(leg, target);
    target_hip = kin_.clamp_to_workspace(leg, target_hip);
    return kin_.hip_to_body(leg, target_hip);
}
```

**摆动相 Bezier 曲线:**

```cpp
Vec3 FootPlanner::swing_trajectory(
    const Vec3& p0,           // 起点
    const Vec3& p3,           // 终点
    double t,                 // [0, 1]
    double step_height) const
{
    // 三阶 Bezier: B(t) = (1-t)³P0 + 3(1-t)²tP1 + 3(1-t)t²P2 + t³P3
    //
    // P1, P2 是控制点，设计为:
    //   P1 = P0 上方 step_height (起飞)
    //   P2 = P3 上方 step_height (落地前)
    // 这保证起飞和落地时垂直速度为零

    Vec3 p1 = p0;
    p1.z += step_height;   // 注意: z轴向上为正 (体坐标系下足端z为负)
    // 如果你的z是向下为正: p1.z -= step_height

    Vec3 p2 = p3;
    p2.z += step_height;

    double u = t;
    double u2 = u * u;
    double u3 = u2 * u;
    double inv = 1.0 - u;
    double inv2 = inv * inv;
    double inv3 = inv2 * inv;

    Vec3 result;
    result.x = inv3*p0.x + 3*inv2*u*p1.x + 3*inv*u2*p2.x + u3*p3.x;
    result.y = inv3*p0.y + 3*inv2*u*p1.y + 3*inv*u2*p2.y + u3*p3.y;
    result.z = inv3*p0.z + 3*inv2*u*p1.z + 3*inv*u2*p2.z + u3*p3.z;

    return result;
}
```

**plan() 主循环:**

```cpp
std::array<FootTarget, kNumLegs> FootPlanner::plan(
    const RobotState& state,
    const LocomotionCommand& cmd,
    const ContactSchedule& schedule)
{
    std::array<FootTarget, kNumLegs> targets;

    for (size_t i = 0; i < kNumLegs; ++i) {
        auto leg = static_cast<LegID>(i);
        auto& mem = memory_[i];
        const auto& leg_state = state.legs[i];
        const bool in_stance = schedule.in_stance[i];

        // 检测相位切换
        if (in_stance && !mem.was_in_stance) {
            // 摆动→支撑: 记录触地位置
            mem.touchdown_pos = leg_state.foot_pos_body;
        }
        if (!in_stance && mem.was_in_stance) {
            // 支撑→摆动: 记录离地位置，计算下一个落脚点
            mem.liftoff_pos = leg_state.foot_pos_body;
            mem.landing_target = compute_landing_target(
                leg,
                kin_.hip_to_body(leg, {0,0,0}),  // hip 在 body 下的位置
                cmd,
                /* gait_params — 需要从外部传入 */ {});
        }

        if (in_stance) {
            // ---- 支撑相: 线性后推 ----
            double progress = schedule.stance_progress[i];
            double T_stance = 1.0; // 需要从 gait params 算

            // 脚相对 body 匀速后移 (等效于 body 匀速前移)
            Vec3 stance_pos = mem.touchdown_pos;
            stance_pos.x -= cmd.vx * progress * T_stance;
            stance_pos.y -= cmd.vy * progress * T_stance;

            targets[i].position = stance_pos;
            targets[i].force_contact = true;
        } else {
            // ---- 摆动相: Bezier 轨迹 ----
            double progress = schedule.swing_progress[i];
            targets[i].position = swing_trajectory(
                mem.liftoff_pos,
                mem.landing_target,
                progress,
                0.04  // step_height, 应该从 gait params 拿
            );
            targets[i].force_contact = false;
        }

        mem.was_in_stance = in_stance;
    }

    return targets;
}
```

## 七、模块 4: 指令滤波器 (CommandFilter)

```cpp
// ======================== command_filter.h ========================

#pragma once
#include "types.h"

namespace quad {

struct FilterConfig {
    double vx_slew{0.8};       // m/s²
    double vy_slew{0.4};
    double yaw_slew{2.0};      // rad/s²
    double height_slew{0.05};  // m/s
    double timeout_s{0.25};
};

class CommandFilter {
public:
    explicit CommandFilter(const FilterConfig& cfg);

    // 接收原始指令
    void set_raw_command(const LocomotionCommand& cmd, double timestamp);

    // 每个控制周期调用，返回滤波后的指令
    LocomotionCommand get_filtered(double dt, double current_time);

    // 是否超时
    bool is_timed_out(double current_time) const;

private:
    FilterConfig cfg_;
    LocomotionCommand raw_{};
    LocomotionCommand filtered_{};
    double last_cmd_time_{0};
```

**update() 主循环实现:**

```cpp
JointCommand LocomotionController::update(double current_time, double dt) {
    // 1. 更新机器人状态 (用关节反馈做FK)
    update_robot_state();

    // 2. 获取滤波后的指令
    LocomotionCommand cmd = cmd_filter_.get_filtered(dt, current_time);
    if (cmd_filter_.is_timed_out(current_time)) {
        cmd = {};  // 归零
        handle_gait_transition(GaitType::STAND, dt);
    }

    // 3. 步态调度 → 接触计划
    ContactSchedule schedule = scheduler_.step(dt);

    // 4. 足端轨迹规划
    auto foot_targets = planner_.plan(state_, cmd, schedule);

    // 5. 逆运动学
    JointCommand output{};
    for (size_t i = 0; i < kNumLegs; ++i) {
        auto leg = static_cast<LegID>(i);
        JointAngles q;

        // 体坐标系 → 髋关节坐标系
        Vec3 target_hip = kinematics_.body_to_hip(leg, foot_targets[i].position);

        // workspace clamp
        target_hip = kinematics_.clamp_to_workspace(leg, target_hip);

        if (kinematics_.inverse_kinematics(leg, target_hip, q)) {
            output.positions[i] = q;
        } else {
            // IK 失败: 用上一次的关节角 (不是跳过!)
            output.positions[i] = joint_fb_[i];
        }
    }

    // 6. 步态切换过渡 (blend)
    if (transitioning_) {
        for (size_t i = 0; i < kNumLegs; ++i) {
            for (size_t j = 0; j < kJointsPerLeg; ++j) {
                double a = transition_start_joints_[i][j];
                double b = output.positions[i][j];
                double t = smooth_step(transition_progress_);
                output.positions[i][j] = a + (b - a) * t;
            }
        }
    }

    return output;
}
```
